021import ball.annotation.ServiceProviderFor;
022import ball.util.ant.taskdefs.AntTask;
023import ball.xml.FluentNode;
024import com.sun.javadoc.ClassDoc;
025import com.sun.javadoc.Tag;
026import com.sun.tools.doclets.Taglet;
027import java.net.URL;
028import java.util.AbstractMap.SimpleEntry;
029import java.util.HashSet;
030import java.util.Map;
031import java.util.Set;
032import java.util.regex.Pattern;
033import lombok.NoArgsConstructor;
034import lombok.ToString;
035import org.apache.tools.ant.ComponentHelper;
036import org.apache.tools.ant.IntrospectionHelper;
037import org.apache.tools.ant.Project;
038import org.apache.tools.ant.Task;
039import org.apache.tools.ant.taskdefs.Antlib;
040import org.w3c.dom.Node;
041
042import static org.apache.commons.lang3.StringUtils.isNotEmpty;
043import static org.apache.commons.lang3.StringUtils.repeat;
044import static org.apache.tools.ant.MagicNames.ANTLIB_PREFIX;
045
046/**
047 * Inline {@link Taglet} to document {@link.uri http://ant.apache.org/ Ant}
048 * {@link Task}s.
049 *
050 * @author {@link.uri mailto:ball@hcf.dev Allen D. Ball}
051 */
052@TagletName("ant.task")
053@ServiceProviderFor({ Taglet.class })
054@NoArgsConstructor @ToString
055public class AntTaskTaglet extends AbstractInlineTaglet implements SunToolsInternalToolkitTaglet {
056    private static final AntTaskTaglet INSTANCE = new AntTaskTaglet();
057
058    public static void register(Map<Object,Object> map) {
059        register(map, INSTANCE);
060    }
061
062    private static final String NO = "no";
063    private static final String YES = "yes";
064
065    private static final String INDENTATION = "  ";
066
067    private static final String DOCUMENTED = "DOCUMENTED";
068
069    @Override
070    public FluentNode toNode(Tag tag) throws Throwable {
071        ClassDoc doc = null;
072        String name = tag.text().trim();
073
074        if (isNotEmpty(name)) {
075            doc = getClassDocFor(tag, name);
076        } else {
077            doc = containingClass(tag);
078        }
079
080        Class<?> type = getClassFor(doc);
081
082        if (! Task.class.isAssignableFrom(type)) {
083            throw new IllegalArgumentException(type.getCanonicalName()
084                                               + " is not a subclass of " + Task.class.getCanonicalName());
085        }
086
087        String template =
088            render(template(tag, type), INDENTATION.length())
089            .replaceAll(Pattern.quote(DOCUMENTED + "=\"\""), "...");
090
091        return div(attr("class", "block"), pre("xml", template));
092    }
093
094    private FluentNode template(Tag tag, Class<?> type) {
095        String name = null;
096
097        if (name == null) {
098            Project project = new Project();
099            String pkg = type.getPackage().getName();
100
101            while (pkg != null) {
102                URL url = type.getResource("/" + String.join("/", pkg.split(Pattern.quote("."))) + "/antlib.xml");
103
104                if (url != null) {
105                    try {
106                        Antlib.createAntlib(project, url, ANTLIB_PREFIX + pkg)
107                            .execute();
108                        break;
109                    } catch (Exception exception) {
110                    }
111                }
112
113                int index = pkg.lastIndexOf(".");
114
115                if (! (index < 0)) {
116                    pkg = pkg.substring(0, index);
117                } else {
118                    pkg = null;
119                }
120            }
121
122            ComponentHelper helper = ComponentHelper.getComponentHelper(project);
123
124            name =
125                helper.getTaskDefinitions().entrySet().stream()
126                .filter(t -> t.getValue().equals(type))
127                .map(t -> t.getKey())
128                .findFirst().orElse(null);
129        }
130
131        if (name == null) {
132            AntTask annotation = type.getAnnotation(AntTask.class);
133
134            name = (annotation != null) ? annotation.value() : null;
135        }
136
137        if (name == null) {
138            name = type.getSimpleName();
139        }
140
141        return type(0, new HashSet<>(), tag, new SimpleEntry<>(name, type));
142    }
143
144    private FluentNode type(int depth, Set<Map.Entry<?,?>> set, Tag tag, Map.Entry<String,Class<?>> entry) {
145        IntrospectionHelper helper = IntrospectionHelper.getHelper(entry.getValue());
146        FluentNode node = element(entry.getKey());
147
148        if (set.add(entry)
149            && (! entry.getValue().getName().startsWith(Task.class.getPackage().getName()))) {
150            node
151                .add(attributes(tag, helper))
152                .add(content(depth + 1, set, tag, helper));
153
154            if (helper.supportsCharacters()) {
155                String content = "... text ...";
156
157                if (node.hasChildNodes()) {
158                    content = "\n" + repeat(INDENTATION, depth + 1) + content + "\n";
159                }
160
161                node.add(text(content));
162            }
163        } else {
164            node.add(attr(DOCUMENTED));
165        }
166
167        return node;
168    }
169
170    private Node[] attributes(Tag tag, IntrospectionHelper helper) {
171        Node[] array =
172            helper.getAttributeMap().entrySet().stream()
173            .map(t -> attr(t.getKey(), t.getValue().getSimpleName()))
174            .toArray(Node[]::new);
175
176        return array;
177    }
178
179    private FluentNode content(int depth, Set<Map.Entry<?,?>> set,
180                               Tag tag, IntrospectionHelper helper) {
181        return fragment(helper.getNestedElementMap().entrySet().stream()
182                        .map(t -> type(depth, set, tag, t)));
183    }
184}
